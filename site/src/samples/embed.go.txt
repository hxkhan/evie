package main

import (
	"fmt"

	"github.com/hxkhan/evie"
)

func main() {
	eng := evie.New()
	defer eng.Close()

	_, err := eng.Eval(`package main
fn greet(name) => "Hello, " + name + "!"`)
	if err != nil {
		panic(err)
	}

	v, err := eng.Call("main.greet", "Gopher")
	if err != nil {
		panic(err)
	}
	fmt.Println(v) // Hello, Gopher!
}
