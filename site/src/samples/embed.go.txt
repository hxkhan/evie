package main

import (
	"fmt"

	"github.com/hxkhan/evie"
	"github.com/hxkhan/evie/vm"
)

func main() {
	// a package of Go functions that scripts can import
	api := vm.NewPackage("api")
	api.Native("user", vm.Func1(func(id int) string {
		return fmt.Sprintf("user-%d", id)
	}))

	eng := evie.New(evie.Options{Packages: []*vm.Package{api}})
	defer eng.Close()

	_, err := eng.Eval(`package main imports("api")
fn greet(id) => "Hi, " + api.user(id).toUpper()`)
	if err != nil {
		panic(err)
	}

	v, _ := eng.Call("main.greet", 7)
	fmt.Println(v) // Hi, USER-7
}
